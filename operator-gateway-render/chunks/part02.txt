  for (const [sessionId, session] of sessions) {
    if (now - session.lastSeenAt > 24 * 60 * 60_000) sessions.delete(sessionId);
  }
  for (const [jobId, job] of jobs) {
    if (job.expiresAt <= now && !job.settled) {
      job.settled = true;
      clearTimeout(job.timeout);
      job.resolve({ ok: false, error: 'Device job timeout' });
      jobs.delete(jobId);
    }
  }
}

function toolError(message) {
  return { content: [{ type: 'text', text: message }], isError: true };
}

function toolSuccess(data, text = null) {
  return {
    content: [{ type: 'text', text: text || JSON.stringify(data) }],
    structuredContent: data,
  };
}

function serialiseJob(jobId, job) {
  return {
    jobId,
    tool: job.tool,
    arguments: job.arguments,
    createdAt: new Date(job.createdAt).toISOString(),
    expiresAt: new Date(job.expiresAt).toISOString(),
  };
}

function takeQueuedJob(deviceId) {
  for (const [jobId, job] of jobs) {
    if (job.deviceId === deviceId && !job.dispatched && !job.settled && job.expiresAt > Date.now()) {
      job.dispatched = true;
      return { jobId, job };
    }
  }
  return null;
}

function wakePoller(deviceId) {
  const waiters = pollWaiters.get(deviceId);
  if (!waiters?.size) return;
  const queued = takeQueuedJob(deviceId);
  if (!queued) return;
  const waiter = waiters.values().next().value;
  waiters.delete(waiter);
  if (!waiters.size) pollWaiters.delete(deviceId);
  waiter.finish(200, { ok: true, job: serialiseJob(queued.jobId, queued.job) });
}

function enqueueToolJob(deviceId, tool, args) {
  const jobId = id('job_');
  const createdAt = Date.now();
  return new Promise((resolve) => {
    const job = {
      deviceId,
      tool,
      arguments: args || {},
      createdAt,
      expiresAt: createdAt + JOB_TIMEOUT_MS,
      dispatched: false,
      settled: false,
      resolve,
      timeout: null,
    };
    job.timeout = setTimeout(() => {
      if (job.settled) return;
      job.settled = true;
      jobs.delete(jobId);
      resolve({ ok: false, error: 'Device job timeout' });
    }, JOB_TIMEOUT_MS);
    jobs.set(jobId, job);
    wakePoller(deviceId);
  });
}
