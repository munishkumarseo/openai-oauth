or(`Tool not found: ${name}`);
  }

  const execution = await enqueueToolJob(session.deviceId, name, args || {});
  if (!execution?.ok) return toolError(String(execution?.error || 'Extension tool failed'));
  if (name === 'capture_current_view') return screenshotToolResult(execution.result);
  return toolSuccess(execution.result ?? {});
}

async function handleMcp(req, res, body) {
  const incomingSession = req.headers['mcp-session-id'];
  const sessionId = typeof incomingSession === 'string' && incomingSession ? incomingSession : id('mcp_');
  let session = sessions.get(sessionId);
  if (!session) {
    session = { id: sessionId, deviceId: null, createdAt: Date.now() };
    sessions.set(sessionId, session);
  }
  const headers = { 'mcp-session-id': sessionId };
  const jsonrpc = body?.jsonrpc;
  const method = body?.method;
  const requestId = body?.id;
  if (jsonrpc !== '2.0' || typeof method !== 'string') {
    return send(res, 400, { jsonrpc: '2.0', id: requestId ?? null, error: { code: -32600, message: 'Invalid Request' } }, headers);
  }
  if (method === 'notifications/initialized' || method === 'notifications/cancelled') return send(res, 202, null, headers);

  let result;
  let error;
  try {
    switch (method) {
      case 'initialize': {
        const supported = ['2025-06-18', '2025-03-26', '2024-11-05'];
        const requested = body?.params?.protocolVersion;
        result = {
          protocolVersion: supported.includes(requested) ? requested : supported[0],
          serverInfo: { name: SERVICE, version: VERSION },
          capabilities: { tools: { listChanged: false } },
        };
        break;
      }
      case 'ping': result = {}; break;
      case 'tools/list': result = { tools: toolsList() }; break;
      case 'tools/call': result = await handleToolCall(session, body?.params?.name, body?.params?.arguments || {}); break;
      case 'prompts/list': result = { prompts: [] }; break;
      case 'resources/list': result = { resources: [] }; break;
      default: error = { code: -32601, message: `Method not found: ${method}` };
    }
  } catch (err) {
    error = { code: -32603, message: 'Internal server error' };
    console.error('[mcp_error]', err?.message || err);
  }
  const response = { jsonrpc: '2.0', id: requestId };
  if (error) response.error = error; else response.result = result;
  return send(res, 200, response, headers);
}
