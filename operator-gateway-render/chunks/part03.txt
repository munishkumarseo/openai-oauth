      annotations: readOnly,
    },
  ];
}

function screenshotToolResult(extensionResult) {
  const dataUrl = extensionResult?.dataUrl;
  if (typeof dataUrl !== 'string') return toolError('Invalid screenshot result');
  let mimeType;
  let base64;
  if (dataUrl.startsWith('data:image/jpeg;base64,')) {
    mimeType = 'image/jpeg'; base64 = dataUrl.slice('data:image/jpeg;base64,'.length);
  } else if (dataUrl.startsWith('data:image/png;base64,')) {
    mimeType = 'image/png'; base64 = dataUrl.slice('data:image/png;base64,'.length);
  } else return toolError('Invalid screenshot format');
  if (!/^[A-Za-z0-9+/]*={0,2}$/.test(base64) || base64.length > MAX_SCREENSHOT_BASE64) {
    return toolError('Invalid or oversized screenshot data');
  }
  const metadata = { ...extensionResult };
  delete metadata.dataUrl;
  return {
    content: [
      { type: 'text', text: JSON.stringify(metadata) },
      { type: 'image', data: base64, mimeType },
    ],
    structuredContent: metadata,
  };
}

async function handleToolCall(session, name, args) {
  if (!session.deviceId) return toolError('This connector URL is not linked to an extension device. Generate a fresh MCP connection URL in the extension and recreate the custom app.');
  const device = devices.get(session.deviceId);
  if (!isOnline(device)) return toolError('Connected extension is offline. Open the extension and keep companion mode enabled.');
  if (!['get_extension_status', 'list_browser_tabs', 'inspect_current_page', 'capture_current_view'].includes(name)) {
    return toolError(`Tool not found: ${name}`);
  }
  const execution = await enqueueToolJob(session.deviceId, name, args || {});
  if (!execution?.ok) return toolError(String(execution?.error || 'Extension tool failed'));
  if (name === 'capture_current_view') return screenshotToolResult(execution.result);
  return toolSuccess(execution.result ?? {});
}

function connectorDeviceFromPath(pathname) {
  const match = pathname.match(/^\/mcp\/([^/]+)$/);
  if (!match) return null;
  return verifyConnectorToken(decodeURIComponent(match[1]));
}

async function handleMcp(req, res, body, connectorInfo) {
  const incomingSession = req.headers['mcp-session-id'];
  const sessionId = typeof incomingSession === 'string' && incomingSession ? incomingSession : id('mcp_');
  let session = sessions.get(sessionId);
  if (!session) {
    session = {
      id: sessionId,
      deviceId: connectorInfo?.deviceId || null,
      connectorExpiresAt: connectorInfo?.expiresAt || null,
      createdAt: Date.now(),
      lastSeenAt: Date.now(),
    };
    sessions.set(sessionId, session);
  } else {
    session.lastSeenAt = Date.now();
    if (connectorInfo?.deviceId) {
      session.deviceId = connectorInfo.deviceId;
      session.connectorExpiresAt = connectorInfo.expiresAt;
    }
  }

  const headers = { 'mcp-session-id': sessionId };
  const jsonrpc = body?.jsonrpc;
  const method = body?.method;
  const requestId = body?.id;
  if (jsonrpc !== '2.0' || typeof method !== 'string') {
    return send(res, 400, { jsonrpc: '2.0', id: requestId ?? null, error: { code: -32600, message: 'Invalid Request' } }, headers);
  }
  if (method === 'notifications/initialized' || method === 'notifications/cancelled') return send(res, 202, null, headers);

  let result;
  let error;
  try {
    switch (method) {
      case 'initialize': {
        const supported = ['2025-06-18', '2025-03-26', '2024-11-05'];
        const requested = body?.params?.protocolVersion;
        result = {
          protocolVersion: supported.includes(requested) ? requested : supported[0],
          serverInfo: { name: SERVICE, version: VERSION },
          capabilities: { tools: { listChanged: false } },
        };
        break;
      }
      case 'ping': result = {}; break;
      case 'tools/list': result = { tools: toolsList() }; break;
      case 'tools/call': result = await handleToolCall(session, body?.params?.name, body?.params?.arguments || {}); break;
      case 'prompts/list': result = { prompts: [] }; break;
      case 'resources/list': result = { resources: [] }; break;
      default: error = { code: -32601, message: `Method not found: ${method}` };
    }
  } catch (err) {
    error = { code: -32603, message: 'Internal server error' };
    console.error('[mcp_error]', err?.message || err);
  }
  const response = { jsonrpc: '2.0', id: requestId };
  if (error) response.error = error; else response.result = result;
  return send(res, 200, response, headers);
}

function publicOrigin(req) {
  const forwardedProto = String(req.headers['x-forwarded-proto'] || 'https').split(',')[0].trim();
  const host = String(req.headers['x-forwarded-host'] || req.headers.host || '').split(',')[0].trim();
  return `${forwardedProto}://${host}`;
}
