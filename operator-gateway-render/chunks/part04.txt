const server = http.createServer(async (req, res) => {
  const url = new URL(req.url, `http://${req.headers.host || 'localhost'}`);
  const path = url.pathname;
  try {
    if (req.method === 'OPTIONS') return send(res, 204);
    if (req.method === 'GET' && path === '/') {
      return send(res, 200, `AI Browser Operator Gateway ${VERSION}\nGenerate a signed MCP connection URL in the extension.\nAll exposed tools are read-only.`);
    }
    if (req.method === 'GET' && path === '/health') {
      cleanup();
      const onlineDevices = [...devices.values()].filter(isOnline).length;
      return send(res, 200, { ok: true, service: SERVICE, version: VERSION, devices: devices.size, onlineDevices, sessions: sessions.size, pendingJobs: jobs.size, time: nowIso() });
    }

    const connectorInfo = connectorDeviceFromPath(path);
    const isMcpPath = path === '/mcp' || Boolean(connectorInfo);
    if (/^\/mcp\//.test(path) && !connectorInfo) return send(res, 401, { ok: false, error: 'Invalid or expired MCP connection URL' });
    if (isMcpPath && req.method === 'GET') return send(res, 405, 'Use POST for MCP Streamable HTTP', { allow: 'POST,DELETE' });
    if (isMcpPath && req.method === 'DELETE') {
      const sessionId = req.headers['mcp-session-id'];
      if (typeof sessionId === 'string') sessions.delete(sessionId);
      return send(res, 204);
    }

    if (req.method === 'POST' && path === '/api/device/register') {
      const body = await readJson(req);
      if (!isDeviceId(body.deviceId)) throw Object.assign(new Error('Invalid deviceId format'), { status: 400 });
      if (!isDeviceSecret(body.deviceSecret)) throw Object.assign(new Error('Invalid deviceSecret format'), { status: 400 });
      const secretHash = sha256(body.deviceSecret);
      const existing = devices.get(body.deviceId);
      if (existing && existing.secretHash !== secretHash) throw Object.assign(new Error('Device authentication failed'), { status: 401 });
      const device = existing || { secretHash, lastSeenAt: 0 };
      device.lastSeenAt = Date.now();
      devices.set(body.deviceId, device);
      const pairedSessionCount = [...sessions.values()].filter((session) => session.deviceId === body.deviceId).length;
      return send(res, 200, { ok: true, gatewayVersion: VERSION, device: { deviceId: body.deviceId, online: true, lastSeenAt: nowIso(), pairedSessionCount } });
    }

    if (req.method === 'POST' && path === '/api/pair/start') {
      const body = await readJson(req);
      const { deviceId } = authDevice(body);
      const { token, expiresAt } = issueConnectorToken(deviceId);
      const connectorUrl = `${publicOrigin(req)}/mcp/${encodeURIComponent(token)}`;
      return send(res, 200, {
        ok: true,
        connectorUrl,
        expiresAt: new Date(expiresAt).toISOString(),
        instructions: 'Create a new ChatGPT custom app using this complete Server URL. No pairing command is required.',
      });
    }

    if (req.method === 'POST' && path === '/api/jobs/next') {
      const body = await readJson(req);
      const { deviceId } = authDevice(body);
      const queued = takeQueuedJob(deviceId);
      if (queued) return send(res, 200, { ok: true, job: serialiseJob(queued.jobId, queued.job) });

      let finished = false;
      const waiter = { finish: null };
      const waiters = pollWaiters.get(deviceId) || new Set();
      const finish = (status, payload) => {
        if (finished || res.writableEnded) return;
        finished = true;
        clearTimeout(timer);
        waiters.delete(waiter);
        if (!waiters.size) pollWaiters.delete(deviceId);
        send(res, status, payload);
      };
      waiter.finish = finish;
      waiters.add(waiter);
      pollWaiters.set(deviceId, waiters);
      const timer = setTimeout(() => finish(204, null), POLL_TIMEOUT_MS);
      req.on('close', () => {
        if (!finished) {
          finished = true;
          clearTimeout(timer);
          waiters.delete(waiter);
          if (!waiters.size) pollWaiters.delete(deviceId);
        }
      });
      return;
    }

    if (req.method === 'POST' && path === '/api/jobs/result') {
      const body = await readJson(req);
      const { deviceId } = authDevice(body);
      const job = jobs.get(body.jobId);
      if (!job || job.deviceId !== deviceId) throw Object.assign(new Error('Job not found or unauthorized'), { status: 404 });
      if (job.settled) return send(res, 200, { ok: true, duplicate: true });
      job.settled = true;
      clearTimeout(job.timeout);
      jobs.delete(body.jobId);
      job.resolve(body.ok === true ? { ok: true, result: body.result } : { ok: false, error: String(body.error || 'Extension tool failed').slice(0, 2000) });
      return send(res, 200, { ok: true });
    }

    if (req.method === 'POST' && isMcpPath) {
      const body = await readJson(req);
      return handleMcp(req, res, body, connectorInfo);
    }

    return send(res, 404, { ok: false, error: 'Not found' });
  } catch (err) {
    const status = Number(err?.status) || 500;
    if (status >= 500) console.error('[request_error]', err?.message || err);
    return send(res, status, { ok: false, error: status >= 500 ? 'Internal server error' : err.message });
  }
});

server.requestTimeout = 45_000;
server.headersTimeout = 50_000;
server.keepAliveTimeout = 5_000;
server.listen(PORT, HOST, () => console.log(`[startup] ${SERVICE} ${VERSION} listening on ${HOST}:${PORT}`));
